cleanup() {
  "$node_bin" "$project_dir/dist/cli/main.js" ops kill-engage \
    --reason "cycle stopped or completed" \
    --actor ozgur >/dev/null 2>&1 || true

  if [[ -n "$cycle_dir" ]]; then
    rm -rf "$cycle_dir"
  fi
}

trap 'exit 130' INT
trap 'exit 143' TERM
trap cleanup EXIT

  --output "$clearance_path" \
  --minutes 30 \
  --actor ozgur \
  --reason "bounded Marx.finance autonomous cycle"

"$node_bin" "$project_dir/dist/cli/main.js" ops kill-clear \
  --clearance "$clearance_path"

  --autonomous \
  --live-read

  --dry-run=false \
  --real-model=false \
  --limit 5 \
  --actions 5
